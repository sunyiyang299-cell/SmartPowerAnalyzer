"""Single-phase sinusoidal steady-state power from nonnegative RMS inputs."""

from dataclasses import dataclass

import numpy as np

from smartpower.analysis.power_factor import analyze_power_factor
from smartpower.analysis.statistics import AnalysisError, Statistics, numeric_samples, summarize

POWER_MODEL = "仅适用于单相正弦稳态交流：U、I 为有效值，PF 为 0～1 的非负功率因数。"
POWER_LIMITATION = "无功功率仅为绝对值，不能判定感性/容性或超前/滞后；不适用于三相或非正弦谐波分析。"


@dataclass(frozen=True)
class PowerAnalysis:
    active: np.ndarray
    apparent: np.ndarray
    reactive_abs: np.ndarray
    active_statistics: Statistics
    apparent_statistics: Statistics
    reactive_abs_statistics: Statistics
    power_factor_statistics: Statistics


def analyze_power(voltage, current, power_factor) -> PowerAnalysis:
    u = numeric_samples(voltage, "电压有效值")
    i = numeric_samples(current, "电流有效值")
    pf = numeric_samples(power_factor, "功率因数", upper=1.0)
    if not (len(u) == len(i) == len(pf)):
        raise AnalysisError("电压、电流与功率因数采样数量必须一致，不能广播计算。")
    try:
        with np.errstate(over="raise", invalid="raise", under="ignore"):
            apparent = u * i
            active = apparent * pf
            # (1-pf)*(1+pf) is algebraically 1-pf**2, with less
            # cancellation near unity. Invalid PF is rejected before this.
            reactive = apparent * np.sqrt(np.maximum(0.0, (1.0 - pf) * (1.0 + pf)))
    except FloatingPointError as exc:
        raise AnalysisError("功率结果超出 float64 可表示范围，请检查单位与输入数量级。") from exc
    if not all(np.isfinite(a).all() for a in (active, apparent, reactive)):
        raise AnalysisError("功率结果包含非有限值，已拒绝本次计算。")
    # A positive physical result rounded down to zero is unrepresentable,
    # not a zero-load measurement. Exact zeros (U/I=0, PF=0 or PF=1) remain valid.
    if (np.any((u > 0) & (i > 0) & (apparent == 0))
            or np.any((apparent > 0) & (pf > 0) & (active == 0))
            or np.any((apparent > 0) & (pf < 1) & (reactive == 0))):
        raise AnalysisError("非零功率结果小于 float64 可表示范围，已拒绝将其当作零值输出。")
    active_stats = summarize(active, "有功功率", "W")
    apparent_stats = summarize(apparent, "视在功率", "VA")
    reactive_stats = summarize(reactive, "无功功率绝对值 |Q|", "var")
    pf_stats = analyze_power_factor(pf)
    for array in (active, apparent, reactive):
        array.setflags(write=False)
    return PowerAnalysis(active, apparent, reactive, active_stats, apparent_stats, reactive_stats, pf_stats)

"""Validated float64 statistics with population (ddof=0) deviation."""

from dataclasses import dataclass

import numpy as np

from smartpower.data.validation import DataError


class AnalysisError(DataError):
    """Invalid analysis input or an unrepresentable numerical result."""


@dataclass(frozen=True)
class Statistics:
    quantity: str
    unit: str
    count: int
    mean: float
    maximum: float
    minimum: float
    population_std: float


def numeric_samples(values, quantity: str, upper: float | None = None) -> np.ndarray:
    """Accept one-dimensional real numeric samples; never coerce text/bools."""
    try:
        raw = np.asarray(values)
        if raw.ndim != 1:
            raise AnalysisError(f"{quantity}必须是一维数值序列。")
        if raw.size == 0:
            raise AnalysisError(f"{quantity}数据集为空，无法计算。")
        if raw.dtype.kind not in "iuf" or (
            isinstance(values, (list, tuple)) and any(isinstance(v, (bool, np.bool_)) for v in values)
        ):
            raise AnalysisError(f"{quantity}类型错误：仅接受实数，不接受文本、布尔值或复数。")
        with np.errstate(over="raise", invalid="raise"):
            samples = raw.astype(np.float64, copy=True)
    except AnalysisError:
        raise
    except (TypeError, ValueError, OverflowError, FloatingPointError) as exc:
        raise AnalysisError(f"{quantity}数据类型或数值范围无效。") from exc
    if not np.isfinite(samples).all():
        raise AnalysisError(f"{quantity}含 NaN 或 Infinity，无法计算。")
    if np.any(samples < 0):
        raise AnalysisError(f"{quantity}不得为负数。")
    if upper is not None and np.any(samples > upper):
        raise AnalysisError(f"{quantity}必须位于 0～{upper:g}（含端点）。")
    return samples


def summarize(values, quantity: str, unit: str, upper: float | None = None) -> Statistics:
    samples = numeric_samples(values, quantity, upper)
    maximum = float(np.max(samples))
    minimum = float(np.min(samples))
    # Subtract before scaling: dividing close large samples by their maximum
    # destroys representable differences. Nonnegative inputs make the range
    # finite; scaling that range also avoids overflow in squared deviations.
    spread = maximum - minimum
    if spread == 0:
        mean, deviation = minimum, 0.0
    else:
        with np.errstate(over="raise", invalid="raise", under="ignore"):
            normalized = (samples - minimum) / spread
            mean = float(minimum + np.mean(normalized, dtype=np.float64) * spread)
            deviation = float(np.std(normalized, dtype=np.float64, ddof=0) * spread)
    if not np.isfinite([mean, deviation]).all():
        raise AnalysisError(f"{quantity}统计结果超出 float64 数值范围。")
    return Statistics(quantity, unit, int(samples.size), mean, maximum, minimum, deviation)

def detect_anomalies(dataset, thresholds: Thresholds, dataset_id=None):
    if not isinstance(dataset, Dataset) or not isinstance(dataset.frame, pd.DataFrame):
        raise DataError("检测输入必须是 V0.2 数据集对象及其数据表。")
    if not isinstance(thresholds, Thresholds):
        raise DataError("请提供有效的用户阈值配置。")
    frame = validate_frame(dataset.frame)
    u, i = frame["电压"].to_numpy(), frame["电流"].to_numpy()
    times = frame["时间"].to_numpy()
    events = []
    for mask, parameter, values, kind, limit, unit in (
        (u < thresholds.voltage_lower, "电压", u, "电压偏低", thresholds.voltage_lower, "V"),
        (u > thresholds.voltage_upper, "电压", u, "电压偏高", thresholds.voltage_upper, "V"),
        (i > thresholds.current_upper, "电流", i, "电流偏高", thresholds.current_upper, "A"),
    ):
        events.extend(AnomalyEvent(dataset_id, int(row), times[row], parameter, float(values[row]),
                                   kind, float(limit), unit, dataset.source)
                      for row in np.flatnonzero(mask))
    events.sort(key=lambda event: (event.row_index, event.kind))
    return DetectionResult(dataset_id, dataset.name, dataset.source, dataset.simulated, len(frame),
                           frame_fingerprint(frame), thresholds, tuple(events),
                           len({event.row_index for event in events}))

@dataclass(frozen=True)
class Thresholds:
    voltage_lower: float
    voltage_upper: float
    current_upper: float

    def __post_init__(self):
        for name, value in (("电压下限", self.voltage_lower), ("电压上限", self.voltage_upper),
                            ("电流上限", self.current_upper)):
            if isinstance(value, bool) or not isinstance(value, Real) or not math.isfinite(value) or value < 0:
                raise DataError(f"{name}必须是非负有限数值，不允许 NaN 或 Infinity。")
        if self.voltage_lower >= self.voltage_upper:
            raise DataError("电压下限必须小于电压上限。")
        if self.current_upper <= 0:
            raise DataError("电流上限必须大于 0。")

def create_application(argv: list[str] | None = None) -> QApplication:
    app = QApplication.instance()
    if app is None:
        app = QApplication(sys.argv if argv is None else argv)
    app.setApplicationName(APP_NAME)
    app.setApplicationVersion(APP_VERSION)
    app.setWindowIcon(QIcon(str(resource_path("app.ico"))))
    app.setOrganizationName(ORGANIZATION)
    app.setStyle("Fusion")
    app.setFont(QFont("Microsoft YaHei UI", 10))
    app.setStyleSheet(LIGHT_THEME)
    if not hasattr(app, "_chinese_translator"):
        translator = QTranslator(app)
        if translator.load("qtbase_zh_CN", QLibraryInfo.path(QLibraryInfo.LibraryPath.TranslationsPath)):
            app.installTranslator(translator)
        app._chinese_translator = translator
    return app


def main() -> int:
    arguments = sys.argv[1:]
    if arguments and arguments[0] in ("--release-check", "--release-reopen"):
        if len(arguments) != 2:
            return 2
        from smartpower.release_check import run_release_check
        return run_release_check(arguments[1], reopen=arguments[0] == "--release-reopen")
    app = create_application()
    window = MainWindow()
    # Show inside the active event loop so an immediate native close cannot
    # occur before QApplication can process its quit request.
    QTimer.singleShot(0, window.show)
    return app.exec()

"""Clearly identified demo data, never equipment measurements."""

import os

import tempfile

import numpy as np

from smartpower.data.importer import Dataset
from smartpower.data.validation import DataError, SIMULATED_SOURCE, SOURCE_COLUMN, validate_frame


def generate_demo(count: int = 1000, seed: int | None = 42) -> Dataset:
    if not isinstance(count, int) or isinstance(count, bool) or count <= 0:
        raise DataError("模拟数据条数必须为正整数。")
    rng = np.random.default_rng(seed)
    frame = pd.DataFrame({
        "时间": pd.date_range("2026-01-01 00:00:00", periods=count, freq="s"),
        "电压": rng.uniform(218, 242, count).round(3),
        "电流": rng.uniform(0.5, 25, count).round(3),
        "功率因数": rng.uniform(0.85, 1, count).round(4),
    })
    return Dataset("单相交流演示（模拟数据）", validate_frame(frame), simulated=True)


def export_demo(dataset: Dataset, path: str | Path, *, overwrite: bool = False) -> None:
    """Publish atomically; existing files require explicit overwrite consent."""
    if not dataset.simulated:
        raise DataError("此入口仅允许导出明确标记的模拟数据。")
    path = Path(path)
    frame = validate_frame(dataset.frame)
    frame[SOURCE_COLUMN] = SIMULATED_SOURCE
    temporary = None
    try:
        if path.exists() and not overwrite:
            raise DataError("目标文件已存在，请确认覆盖后重试。")
        with tempfile.NamedTemporaryFile(dir=path.parent, suffix=".tmp", delete=False) as handle:
            temporary = Path(handle.name)
        frame.to_csv(temporary, index=False, encoding="utf-8-sig")
        if overwrite:
            os.replace(temporary, path)
        elif os.name == "nt":
            os.rename(temporary, path)  # Windows refuses an existing destination.
        else:
            os.link(temporary, path)  # Exclusive publication also closes races.
            temporary.unlink()
    except OSError as exc:
        raise DataError(f"导出失败：{exc}。原目标文件未被替换，请检查路径、权限或磁盘空间。") from exc
    finally:
        if temporary is not None:
            temporary.unlink(missing_ok=True)


def main() -> None:
    import argparse
    import sys
    # Redirected output on English Windows can default to cp1252. Use UTF-8
    # explicitly so a successful export never fails while printing its result.
    for stream in (sys.stdout, sys.stderr):
        if hasattr(stream, "reconfigure"):
            stream.reconfigure(encoding="utf-8")
    parser = argparse.ArgumentParser(description="生成 1000 条模拟单相交流数据，非真实设备测量。")
    parser.add_argument("--output", type=Path, default=Path("模拟数据.csv"), help="输出 CSV 路径")
    parser.add_argument("--overwrite", action="store_true", help="明确允许覆盖已有文件")
    args = parser.parse_args()
    if args.output.exists() and not args.overwrite:
        parser.error("目标文件已存在。如需覆盖，请明确添加 --overwrite。")
    try:
        export_demo(generate_demo(), args.output, overwrite=args.overwrite)
    except DataError as exc:
        parser.error(str(exc))
    print(f"已导出 1000 条模拟数据（非真实设备测量）：{args.output}")


if __name__ == "__main__":
    main()

"""Pandas file readers, independent of Qt and persistence."""

from dataclasses import dataclass
from pathlib import Path

import pandas as pd

from smartpower.data.validation import (
    DataError, IMPORTED_SOURCE, REQUIRED_COLUMNS, SIMULATED_SOURCE, SOURCE_COLUMN, validate_frame,
)


@dataclass
class Dataset:
    name: str
    frame: pd.DataFrame
    simulated: bool = False
    warning: str = ""

    @property
    def source(self) -> str:
        return SIMULATED_SOURCE if self.simulated else IMPORTED_SOURCE


def import_file(path: str | Path) -> Dataset:
    path = Path(path)
    if path.suffix.lower() not in (".csv", ".xlsx"):
        raise DataError("仅支持 .csv 和 .xlsx 文件。")
    try:
        if path.stat().st_size == 0:
            raise DataError("文件为空，请选择包含表头和数据的文件。")
        if path.suffix.lower() == ".csv":
            try:
                raw = pd.read_csv(path, encoding="utf-8-sig", dtype=object, header=None,
                                  keep_default_na=False, skip_blank_lines=False)
            except UnicodeDecodeError:
                raw = pd.read_csv(path, encoding="gb18030", dtype=object, header=None,
                                  keep_default_na=False, skip_blank_lines=False)
        else:
            raw = pd.read_excel(path, engine="openpyxl", sheet_name=0, header=None,
                                dtype=object, keep_default_na=False)
    except DataError:
        raise
    except pd.errors.EmptyDataError as exc:
        raise DataError("文件为空或没有可识别的表头。") from exc
    except Exception as exc:
        raise DataError(f"无法读取文件“{path.name}”：{exc}。请检查文件格式、权限及是否损坏。") from exc
    if raw.empty:
        raise DataError("文件没有数据记录。")
    headers = raw.iloc[0].tolist()
    raw = raw.iloc[1:].reset_index(drop=True)
    raw.columns = headers
    frame = validate_frame(raw)
    raw.columns = [str(c).strip().lstrip("\ufeff") for c in raw.columns]
    simulated = SOURCE_COLUMN in raw and raw[SOURCE_COLUMN].astype(str).str.contains("模拟", regex=False).any()
    extra = [c for c in raw.columns if c not in (*REQUIRED_COLUMNS, SOURCE_COLUMN)]
    warning = ""
    if extra:

from dataclasses import dataclass

import numpy as np
import pandas as pd

REQUIRED_COLUMNS = ("时间", "电压", "电流", "功率因数")
SOURCE_COLUMN = "数据来源"
SIMULATED_SOURCE = "模拟数据（非真实设备测量）"
IMPORTED_SOURCE = "导入文件（来源未核验）"


class DataError(ValueError):
    """Actionable user-facing data error."""


@dataclass(frozen=True)
class Issue:
    row: int
    column: str
    reason: str


class ValidationError(DataError):
    def __init__(self, issues: list[Issue], count: int):
        self.issues = issues
        self.count = count
        detail = "\n".join(f"数据第 {i.row} 行 · {i.column}：{i.reason}" for i in issues)
        super().__init__(f"校验失败，共 {count} 处错误（数据行从表头下一行开始计数）。\n{detail}"
                         + ("\n仅展示前 40 处，请修正文件后重新导入。" if count > len(issues) else ""))


def validate_frame(raw: pd.DataFrame) -> pd.DataFrame:
    """Return a normalized copy, never mutate the input or invent values.

    Times are local calendar strings or Excel datetime cells, without timezone.
    Numeric epoch/Excel serial timestamps are rejected to avoid guessing units.
    Extra columns are handled explicitly by the importer.
    """
    if raw.empty:
        raise DataError("文件没有数据记录，请检查是否为空文件或只有表头。")
    frame = raw.copy(deep=False)
    columns = [str(c).strip().lstrip("\ufeff") for c in frame.columns]
    if len(columns) != len(set(columns)):
        raise DataError("存在重复字段名称，请为每一列使用唯一的表头。")
    frame = frame.set_axis(columns, axis=1)
    missing = [c for c in REQUIRED_COLUMNS if c not in columns]
    if missing:
        raise DataError("缺少必需字段：" + "、".join(missing))
    result = pd.DataFrame(index=range(len(frame)))
    issues: list[Issue] = []
    count = 0

    def record(mask, column, reason):
        nonlocal count
        positions = np.flatnonzero(np.asarray(mask, dtype=bool))
        count += len(positions)
        for pos in positions[:max(0, 40 - len(issues))]:
            issues.append(Issue(int(pos) + 1, column, reason))

    for column in REQUIRED_COLUMNS:
        series = frame[column].reset_index(drop=True)
        text = series.astype("string").str.strip()
        missing_mask = series.isna() | text.eq("").fillna(False)
        record(missing_mask, column, "缺失值")
        if column == "时间":
            calendar = text.str.match(
                r"^\d{4}[-/]\d{1,2}[-/]\d{1,2}(?:[ T]\d{1,2}:\d{2}(?::\d{2}(?:\.\d{1,9})?)?)?$"
            ).fillna(False)
            values = pd.to_datetime(text.where(calendar).str.replace("/", "-"),
                                    format="mixed", errors="coerce")
            record(~missing_mask & values.isna(), column,
                   "无效时间，请使用年-月-日 时:分:秒；不接受数字时间戳或时区")
        else:
            # Strings exclude booleans and complex values as well as nonnumeric text.
            if (pd.api.types.is_numeric_dtype(series.dtype)
                    and not pd.api.types.is_bool_dtype(series.dtype)
                    and not pd.api.types.is_complex_dtype(series.dtype)):
                # Preserve native numeric precision, especially nextafter values
                # on strict anomaly thresholds; avoid float -> text -> float.
                values = series.astype(float)
            else:
                values = pd.to_numeric(text, errors="coerce").astype(float)
            invalid = ~np.isfinite(values.to_numpy())
            record(~missing_mask & invalid, column, "必须为有限的合法数值")
            if column == "功率因数":
                record(~missing_mask & ~invalid & ((values < 0) | (values > 1)),
                       column, "必须位于 0～1（含端点）")
            else:
                record(~missing_mask & ~invalid & (values < 0), column, "有效值不得为负数")
        result[column] = values
    if count:
        raise ValidationError(issues, count)
    return result

def representative_indices(values, limit=1800):
    """Preserve bucket extrema for display only, retaining original time mapping."""
    if len(values) <= limit:
        return np.arange(len(values))
    selected = {0, len(values) - 1}
    for bucket in np.array_split(np.arange(len(values)), max(1, (limit - 2) // 2)):
        selected.add(int(bucket[np.argmin(values[bucket])]))
        selected.add(int(bucket[np.argmax(values[bucket])]))
    return np.array(sorted(selected))

"""Atomic report publication; exporters never touch the original target first."""

from contextlib import contextmanager
import os
from pathlib import Path
import tempfile

from smartpower.data.validation import DataError


@contextmanager
def atomic_report(path, suffix, *, overwrite=False):
    target = Path(path)
    if target.suffix.lower() != suffix:
        raise DataError(f"报告文件必须使用 {suffix} 扩展名。")
    if not target.parent.is_dir():
        raise DataError("保存目录不存在，请选择已有目录；未自动创建目录。")
    temporary = None
    try:
        if target.exists() and not overwrite:
            raise DataError("报告文件已存在，请先确认覆盖。")
        with tempfile.NamedTemporaryFile(dir=target.parent, suffix=suffix, delete=False) as stream:
            temporary = Path(stream.name)
        yield temporary
        if not temporary.stat().st_size:
            raise DataError("报告为空，未发布文件。")
        if overwrite:
            os.replace(temporary, target)
        elif os.name == "nt":
            os.rename(temporary, target)
        else:
            os.link(temporary, target)
            temporary.unlink()
        temporary = None
    except DataError:
        raise
    except Exception as exc:
        raise DataError(f"报告导出失败：{exc}。已有文件未被本次失败写入替换。") from exc
    finally:
        if temporary is not None:
            temporary.unlink(missing_ok=True)

def report_font(path=""):
    with FONT_LOCK:
        if not path:
            name = "STSong-Light"
            if name not in pdfmetrics.getRegisteredFontNames():
                pdfmetrics.registerFont(UnicodeCIDFont(name))
            return name
        target = Path(path)
        if not target.is_file() or target.suffix.lower() not in (".ttf", ".ttc"):
            raise DataError("中文字体不可用，请在系统设置中选择具备嵌入许可的 TrueType .ttf/.ttc 文件；或清空路径使用标准中文字体引用。")
        try:
            name = "SPA_" + sha256(target.read_bytes()).hexdigest()[:20]
            if name not in pdfmetrics.getRegisteredFontNames():
                font = TTFont(name, str(target), validate=1)
                # ReportLab rejects restricted embedding in its TrueType parser.
                pdfmetrics.registerFont(font)
            return name
        except Exception as exc:
            raise DataError(f"中文字体无法嵌入：{exc}。请选择具有许可且包含所需中文的 TrueType 字体，或清空路径使用标准引用。") from exc


def ensure_glyphs(font_name, text):
    if font_name == "STSong-Light":
        missing = {c for c in text if ord(c) > 0xFFFF}
    else:
        mapping = pdfmetrics.getFont(font_name).face.charToGlyph
        missing = {c for c in text if not c.isspace() and ord(c) not in mapping}
    if missing:
        codes = "、".join(f"U+{ord(c):04X}" for c in sorted(missing)[:12])

class DeferredEventTable(Flowable):
    """Materialize at most one bounded event batch when Platypus reaches it."""
    BATCH_SIZE = 256

    def __init__(self, events, start, stop, make_table):
        super().__init__()
        self.events, self.start, self.stop = events, start, stop
        self.make_table = make_table
        self._table = None

    def rows(self):
        yield ("原始索引", "采样时间", "异常类型", "采样值", "用户阈值", "单位")
        for index in range(self.start, self.stop):
            e = self.events[index]
            yield (e.row_index, str(e.timestamp).replace("T", " "), e.kind,
                   f"{e.value:.12g}", f"{e.threshold:.12g}", e.unit)

    def _materialize(self):
        if self._table is None:
            self._table = self.make_table(list(self.rows()), [48, 151, 72, 82, 82, 60])
        return self._table

    def wrap(self, width, height):
        return self._materialize().wrapOn(self.canv, width, height)

    def split(self, width, height):
        return self._materialize().splitOn(self.canv, width, height)

    def draw(self):
        self._materialize().drawOn(self.canv, 0, 0)

def frozen_array(values):
    values = np.asarray(values)
    # bytes-backed buffers cannot be made writable again by setflags.
    return np.frombuffer(values.tobytes(), dtype=values.dtype).reshape(values.shape)


def build_report_snapshot(repository, dataset_id, *, title=DEFAULT_TITLE, thresholds=None):
    title = validate_title(title)
    loaded = repository.load(dataset_id)
    if not isinstance(loaded.name, str) or any(ord(c) < 32 for c in loaded.name):
        raise DataError("数据集名称包含控制字符或类型无效，无法生成报告；请先更正名称。")
    dataset = Dataset(loaded.name, validate_frame(loaded.frame), loaded.simulated)
    analysis = analyze_dataset(dataset, "all", dataset_id)
    sample = analysis.samples
    sample = replace(sample, order=frozen_array(sample.order), seconds=frozen_array(sample.seconds),
                     voltage=frozen_array(sample.voltage), current=frozen_array(sample.current))
    power = replace(analysis.power, active=frozen_array(analysis.power.active),
                    apparent=frozen_array(analysis.power.apparent), reactive_abs=frozen_array(analysis.power.reactive_abs))
    analysis = replace(analysis, times=frozen_array(analysis.times), samples=sample, power=power)
    detection = detect_anomalies(dataset, thresholds, dataset_id) if thresholds is not None else None
    return ReportSnapshot(str(uuid4()), APP_NAME, APP_VERSION, title, datetime.now(timezone.utc), analysis,
                          frozen_array(dataset.frame["功率因数"].to_numpy()), thresholds, detection)

def initialize_database(path: str | Path) -> None:
    """Create only schema metadata when explicitly requested by a caller.

    No database is created at application startup. Dataset tables and schema
    migrations belong to subsequent data-management releases.
    """
    connection = sqlite3.connect(path)
    try:
        with connection:
            connection.execute(
                "CREATE TABLE IF NOT EXISTS app_metadata "
                "(key TEXT PRIMARY KEY, value TEXT NOT NULL)"
            )
            connection.execute(
                "INSERT OR IGNORE INTO app_metadata (key, value) VALUES (?, ?)",
                ("schema_version", "1"),
            )
    finally:
        connection.close()

"""Transactional dataset repository. No Qt or GUI state in this module."""

from contextlib import contextmanager
from datetime import datetime
from pathlib import Path
import sqlite3

import pandas as pd

from smartpower.data.importer import Dataset
from smartpower.data.validation import DataError, validate_frame


class StorageError(DataError):
    """Storage errors translated for the UI without hiding the original cause."""


class DatasetRepository:
    def __init__(self, path: str | Path):
        self.path = Path(path)

    @contextmanager
    def _connection(self):
        connection = None
        try:
            self.path.parent.mkdir(parents=True, exist_ok=True)
            connection = sqlite3.connect(self.path, timeout=5)
            connection.execute("PRAGMA foreign_keys=ON")
            with connection:
                connection.execute("CREATE TABLE IF NOT EXISTS app_metadata "
                                   "(key TEXT PRIMARY KEY, value TEXT NOT NULL)")
                row = connection.execute("SELECT value FROM app_metadata WHERE key='schema_version'").fetchone()
                if row and row[0] not in ("1", "2"):
                    raise StorageError("数据库版本不受支持，请使用相应版本的软件，原数据未修改。")
                connection.execute("""CREATE TABLE IF NOT EXISTS datasets (
                    id INTEGER PRIMARY KEY, name TEXT NOT NULL CHECK(length(trim(name))>0),
                    imported_at TEXT NOT NULL, simulated INTEGER NOT NULL CHECK(simulated IN (0,1)),
                    row_count INTEGER NOT NULL CHECK(row_count>0))""")
                connection.execute("""CREATE TABLE IF NOT EXISTS samples (
                    dataset_id INTEGER NOT NULL REFERENCES datasets(id) ON DELETE CASCADE,
                    position INTEGER NOT NULL, sampled_at TEXT NOT NULL,
                    voltage REAL NOT NULL CHECK(voltage>=0), current REAL NOT NULL CHECK(current>=0),
                    power_factor REAL NOT NULL CHECK(power_factor>=0 AND power_factor<=1),
                    PRIMARY KEY(dataset_id, position))""")
                connection.execute("INSERT INTO app_metadata VALUES ('schema_version','2') "
                                   "ON CONFLICT(key) DO UPDATE SET value='2'")
            yield connection
        except StorageError:
            raise
        except (sqlite3.Error, OSError) as exc:
            raise StorageError(f"数据库操作失败：{exc}。本次写入已回滚，当前预览仍可重新保存；请检查路径、权限或磁盘空间。") from exc
        finally:
            if connection is not None:
                connection.close()

    def save(self, dataset: Dataset, name: str) -> int:
        name = name.strip()
        if not name or len(name) > 200:
            raise DataError("数据集名称不能为空，且不能超过 200 个字符。")
        frame = validate_frame(dataset.frame)
        with self._connection() as connection:
            with connection:
                cursor = connection.execute(
                    "INSERT INTO datasets(name,imported_at,simulated,row_count) VALUES (?,?,?,?)",
                    (name, datetime.now().astimezone().isoformat(timespec="seconds"),
                     int(dataset.simulated), len(frame)),
                )
                dataset_id = cursor.lastrowid
                connection.executemany(
                    "INSERT INTO samples VALUES (?,?,?,?,?,?)",
                    ((dataset_id, pos, time.isoformat(), float(voltage), float(current), float(pf))
                     for pos, (time, voltage, current, pf) in
                     enumerate(frame.itertuples(index=False, name=None))),
                )
        return int(dataset_id)

    def list_datasets(self) -> list[dict]:
        with self._connection() as connection:
            connection.row_factory = sqlite3.Row
            return [dict(row) for row in connection.execute(
                "SELECT * FROM datasets ORDER BY id DESC")]

    def load(self, dataset_id: int) -> Dataset:
        with self._connection() as connection:
            with connection:
                connection.execute("BEGIN")
                row = connection.execute("SELECT name,simulated,row_count FROM datasets WHERE id=?",
                                         (dataset_id,)).fetchone()
                if row is None:
                    raise StorageError("指定数据集不存在，可能已被删除，请刷新历史列表。")
                records = connection.execute(
                    "SELECT sampled_at,voltage,current,power_factor FROM samples "
                    "WHERE dataset_id=? ORDER BY position", (dataset_id,)).fetchall()
                if len(records) != row[2]:
                    raise StorageError("数据集记录数与元数据不一致，已停止读取，请保留数据库并检查。")
        try:
            frame = validate_frame(pd.DataFrame(records, columns=["时间", "电压", "电流", "功率因数"]))
        except DataError as exc:
            raise StorageError(f"历史数据校验失败：{exc}") from exc
        return Dataset(row[0], frame, bool(row[1]))

    def delete(self, dataset_id: int) -> None:
        """Caller must obtain user confirmation before invoking this method."""
        with self._connection() as connection:
            with connection:
                cursor = connection.execute("DELETE FROM datasets WHERE id=?", (dataset_id,))
                if cursor.rowcount != 1:
                    raise StorageError("数据集不存在或已删除，请刷新历史列表。")

"""Lazy display of calculated power sequences, without per-row widgets."""

import numpy as np
from PySide6.QtCore import QAbstractTableModel, QModelIndex, Qt


class PowerSeriesModel(QAbstractTableModel):
    HEADERS = ("时间", "有功功率 P（W）", "视在功率 S（VA）", "无功功率绝对值 |Q|（var）")

    def __init__(self, parent=None):
        super().__init__(parent)
        self.result = None

    def set_result(self, result):
        self.beginResetModel()
        self.result = result
        self.endResetModel()

    def rowCount(self, parent=QModelIndex()):
        return 0 if parent.isValid() or self.result is None else self.result.count

    def columnCount(self, parent=QModelIndex()):
        return 0 if parent.isValid() else 4

    def data(self, index, role=Qt.ItemDataRole.DisplayRole):
        if (not index.isValid() or self.result is None or
                not 0 <= index.row() < self.result.count or not 0 <= index.column() < 4):
            return None
        if role == Qt.ItemDataRole.DisplayRole:
            if index.column() == 0:

"""Atomic PNG writing, separate from UI path selection and confirmation."""

import os
from pathlib import Path
import tempfile

from smartpower.data.validation import DataError


def save_png(image, path, *, overwrite=False):
    target = Path(path)
    if target.suffix.lower() != ".png":
        raise DataError("图表文件必须使用 .png 扩展名。")
    temporary = None
    try:
        if target.exists() and not overwrite:
            raise DataError("目标文件已存在，必须先确认覆盖。")
        with tempfile.NamedTemporaryFile(dir=target.parent, suffix=".png", delete=False) as stream:
            temporary = Path(stream.name)
        if not image.save(str(temporary), "PNG"):
            raise DataError("PNG 图像编码或写入失败。")
        if overwrite:
            os.replace(temporary, target)
        else:
            # Exclusive creation prevents an intervening writer being overwritten.
            if os.name == "nt":
                os.rename(temporary, target)  # Windows rename fails if target exists.
            else:
                os.link(temporary, target)
                temporary.unlink()
        temporary = None
    except (OSError, ValueError) as exc:
        raise DataError(f"PNG 导出失败：{exc}。原文件保留，请检查目录和写入权限。") from exc
    finally:
        if temporary is not None:
            temporary.unlink(missing_ok=True)

"""GUI-independent identity and stable time-order preparation."""

from dataclasses import dataclass
from hashlib import sha256

import numpy as np


def timestamp_parts(frame):
    times = frame["时间"].to_numpy()
    seconds = times.astype("datetime64[s]")
    fractions = (times - seconds).astype("timedelta64[ns]").astype("<i8")
    return times, seconds.astype("<i8"), fractions


def frame_fingerprint(frame):
    digest = sha256()
    _, seconds, fractions = timestamp_parts(frame)
    digest.update(seconds.tobytes())
    digest.update(fractions.tobytes())
    digest.update(frame[["电压", "电流", "功率因数"]].to_numpy(dtype="<f8").tobytes())
    return digest.hexdigest()


@dataclass(frozen=True)
class PlotSamples:
    fingerprint: str
    order: np.ndarray
    seconds: np.ndarray
    origin: np.datetime64
    voltage: np.ndarray
    current: np.ndarray


def prepare_samples(frame):
    times, ticks, fractions = timestamp_parts(frame)
    order = np.argsort(times, kind="stable")
    origin = times[order[0]]
    # Relative seconds preserve subsecond precision near modern dates. Avoid int64
    # subtraction overflow for data spanning the whole supported datetime range.
    seconds = ((ticks[order] - ticks[order[0]]).astype(float)
               + (fractions[order] - fractions[order[0]]) / 1e9)
    arrays = [order, seconds, frame["电压"].to_numpy(copy=True), frame["电流"].to_numpy(copy=True)]
    for array in arrays:
        array.setflags(write=False)
    return PlotSamples(frame_fingerprint(frame), order, seconds, origin, arrays[2], arrays[3])

        if self.origin is None:
            return [""] * len(values)
        labels = []
        for value in values:
            try:
                # Calendar formatting via datetime64 avoids system timezone/epoch limits.
                stamp = self.origin.astype("datetime64[ms]") + np.timedelta64(round(value * 1000), "ms")
                labels.append(np.datetime_as_string(stamp, unit="ms").replace("T", "\n")
                              if spacing < 1 else np.datetime_as_string(stamp, unit="s").replace("T", "\n"))
            except (ValueError, OverflowError):
                labels.append("")
        return labels
